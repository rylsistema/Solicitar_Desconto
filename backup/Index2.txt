const brl = new Intl.NumberFormat('pt-BR', {
  style:'currency',
  currency:'BRL'
});

function formatBRL(v){
  return brl.format(Number(v || 0));
}

function parseBRNumber(str){
  if(!str) return 0;

  const s = String(str)
    .replace(/R\$/g,'')
    .replace(/\./g,'')
    .replace(',','.')
    .replace('%','')
    .trim();

  return Number(s) || 0;
}

function clamp(n,a,b){
  return Math.min(Math.max(n,a),b);
}

function showToast(msg){
  const t = document.getElementById('toast');
  t.textContent = msg;
  t.classList.add('show');

  setTimeout(()=>{
    t.classList.remove('show');
  },1800);
}

function adicionarProduto(){

  const original = document.querySelector('.produto');

  const clone = original.cloneNode(true);

  clone.querySelectorAll('input').forEach(input=>{

    if(input.readOnly){
      input.value = input.classList.contains('precoPolitica') || input.classList.contains('difItem')
      ? 'R$ 0,00'
      : '0,00%';

    } else {

      if(input.classList.contains('precoTabela') || input.classList.contains('precoNegociado')){
        input.value = 'R$ 0,00';
      }
      else if(input.classList.contains('descPolitica')){
        input.value = '0,00%';
      }
      else {
        input.value = '';
      }
    }

  });

  document.getElementById('produtosContainer').appendChild(clone);
}

function recalc(){

  let totalPedido = 0;
  let totalDiferenca = 0;

  const produtos = document.querySelectorAll('.produto');

  produtos.forEach(produto=>{

    const qtd = parseBRNumber(produto.querySelector('.qtd').value);
    const precoTabela = parseBRNumber(produto.querySelector('.precoTabela').value);
    const descPolitica = clamp(parseBRNumber(produto.querySelector('.descPolitica').value),0,100);
    const precoNegociado = parseBRNumber(produto.querySelector('.precoNegociado').value);

    const precoPolitica = precoTabela * (1 - (descPolitica / 100));

    let descAMais = 0;

    if(precoPolitica > 0){
      descAMais = (precoPolitica - precoNegociado) / precoPolitica;
    }

    descAMais = Math.max(0, descAMais);

    const totalDesc = descPolitica + (descAMais * 100);

    const difItem = (precoPolitica - precoNegociado) * qtd;

    const totalItem = precoNegociado * qtd;

    totalPedido += totalItem;
    totalDiferenca += difItem;

    produto.querySelector('.precoPolitica').value = formatBRL(precoPolitica);
    produto.querySelector('.descAMais').value = `${(descAMais * 100).toLocaleString('pt-BR',{minimumFractionDigits:2})}%`;
    produto.querySelector('.totalDesconto').value = `${totalDesc.toLocaleString('pt-BR',{minimumFractionDigits:2})}%`;
    produto.querySelector('.difItem').value = formatBRL(difItem);

  });

  document.getElementById('totalPedido').value = formatBRL(totalPedido);
  document.getElementById('difTotal').value = formatBRL(totalDiferenca);

  buildWhatsText();
}

function buildWhatsText(){

  const lines = [];

  const tipoDesconto = document.querySelector('input[name="tipoDesconto"]:checked').value;
  const faturamentoPor = document.querySelector('input[name="faturamentoPor"]:checked').value;

  lines.push('*SOLICITAÇÃO DE DESCONTO*');
  lines.push('');

  lines.push(`• *Tipo de desconto:* ${tipoDesconto}`);
  lines.push(`• *Faturamento por:* ${faturamentoPor}`);

  const codRca = document.getElementById('codRca').value;
  const nomeRca = document.getElementById('nomeRca').value;

  if(codRca || nomeRca){
    lines.push(`• *RCA:* ${codRca} - ${nomeRca}`);
  }

  const codCliente = document.getElementById('codCliente').value;
  const nomeCliente = document.getElementById('nomeCliente').value;

  if(codCliente || nomeCliente){
    lines.push(`• *Cliente:* ${codCliente} - ${nomeCliente}`);
  }

  const prazo = document.getElementById('prazo').value;

  if(prazo){
    lines.push(`• *Prazo:* ${prazo}`);
  }

  lines.push('');
  lines.push('*Produtos*');

  const produtos = document.querySelectorAll('.produto');

  produtos.forEach(p=>{

    const cod = p.querySelector('.codProduto').value || '';
    const nome = p.querySelector('.nomeProduto').value || '';
    const qtd = parseBRNumber(p.querySelector('.qtd').value);

    const precoTabela = parseBRNumber(p.querySelector('.precoTabela').value);
    const descPolitica = clamp(parseBRNumber(p.querySelector('.descPolitica').value),0,100);
    const precoNegociado = parseBRNumber(p.querySelector('.precoNegociado').value);

    const precoPolitica = precoTabela * (1 - (descPolitica / 100));

    let descAMais = 0;

    if(precoPolitica > 0){
      descAMais = (precoPolitica - precoNegociado) / precoPolitica;
    }

    descAMais = Math.max(0, descAMais);

    const totalDesc = descPolitica + (descAMais * 100);
    const difItem = (precoPolitica - precoNegociado) * qtd;

    lines.push('');
    lines.push(`• ${cod} - ${nome}`);
    lines.push(`Qtd: ${qtd}`);
    lines.push(`Preço de Tabela: ${formatBRL(precoTabela)}`);
    lines.push(`% Desconto Política: ${descPolitica.toLocaleString('pt-BR',{minimumFractionDigits:2})}%`);
    lines.push(`Preço da Política: ${formatBRL(precoPolitica)}`);
    lines.push(`Preço Negociado: ${formatBRL(precoNegociado)}`);
    lines.push(`Desc. a Mais: ${(descAMais * 100).toLocaleString('pt-BR',{minimumFractionDigits:2})}%`);
    lines.push(`Total de Desconto: ${totalDesc.toLocaleString('pt-BR',{minimumFractionDigits:2})}%`);
    lines.push(`Diferença Total do Item: ${formatBRL(difItem)}`);

  });

  lines.push('');
  lines.push('*Totais*');
  lines.push(`• *Total valor do pedido:* ${document.getElementById('totalPedido').value}`);
  lines.push(`• *Diferença de preço total:* ${document.getElementById('difTotal').value}`);

  document.getElementById('whatsPreview').value = lines.join('\n');
}

async function copyToClipboard(text){

  if(navigator.clipboard && window.isSecureContext){
    await navigator.clipboard.writeText(text);
    return;
  }

  const ta = document.createElement('textarea');
  ta.value = text;

  document.body.appendChild(ta);
  ta.select();

  document.execCommand('copy');

  document.body.removeChild(ta);
}

document.addEventListener('input', function(e){

  if(
    e.target.classList.contains('qtd') ||
    e.target.classList.contains('precoTabela') ||
    e.target.classList.contains('precoNegociado') ||
    e.target.classList.contains('descPolitica') ||
    e.target.classList.contains('codProduto') ||
    e.target.classList.contains('nomeProduto') ||
    e.target.id === 'codRca' ||
    e.target.id === 'nomeRca' ||
    e.target.id === 'codCliente' ||
    e.target.id === 'nomeCliente' ||
    e.target.id === 'prazo'
  ){
    recalc();
  }

});

document.addEventListener('blur', function(e){

  if(e.target.classList.contains('precoTabela') || e.target.classList.contains('precoNegociado')){
    e.target.value = formatBRL(parseBRNumber(e.target.value));
  }

  if(e.target.classList.contains('descPolitica')){

    const valor = clamp(parseBRNumber(e.target.value),0,100);

    e.target.value = `${valor.toLocaleString('pt-BR',{minimumFractionDigits:2})}%`;
  }

  recalc();

}, true);

document.getElementById('copiar').addEventListener('click', async ()=>{

  try {

    await copyToClipboard(document.getElementById('whatsPreview').value);

    showToast('Texto copiado!');

  } catch(e){
    showToast('Erro ao copiar');
  }

});

document.getElementById('limpar').addEventListener('click', ()=>{
  location.reload();
});

recalc();
